\begin{frame}
  \frametitle{Gini đo mức độ hỗn tạp tại mỗi nút}
  \chaptertag{3}{KIẾN THỨC NỀN TẢNG}
  \begin{tcolorbox}[focusbox,width=0.78\textwidth,top=4mm,bottom=4mm]
    \centering
    {\Large
    $\displaystyle Gini(S)=1-\sum_{k=1}^{K}p_k^2$}
  \end{tcolorbox}
  \vspace{0.45cm}
  \begin{itemize}
    \item $p_k$ là tỷ lệ lớp $k$ trong tập mẫu tại nút.
    \item $Gini=0$ khi tất cả mẫu thuộc cùng một lớp.
    \item CART tìm phép chia làm hai nút con thuần hơn nút cha.
  \end{itemize}
  \note{[Sources] Breiman et al., 1984.\\
  Talk track (0:35): Giải thích trực giác: Gini thấp nghĩa nút gần thuần. Công thức là nền tảng để so sánh các candidate split ở slide tiếp theo.}
\end{frame}

% 8 --------------------------------------------------------------------------
\begin{frame}
  \frametitle{CART chọn split tạo mức giảm Gini lớn nhất}
  \chaptertag{3}{KIẾN THỨC NỀN TẢNG}
  \begin{tcolorbox}[focusbox,width=0.88\textwidth,top=3mm,bottom=3mm]
    \centering
    $\displaystyle
      \Delta G(j,t)=Gini(S)
      -\frac{n_L}{n}Gini(S_L)
      -\frac{n_R}{n}Gini(S_R)$\\[0.35cm]
    $\displaystyle (j^\star,t^\star)=\arg\max_{j,t}\Delta G(j,t)$
  \end{tcolorbox}
  \vspace{0.40cm}
  \begin{itemize}
    \item $j$ là feature; $t$ là threshold.
    \item Gain lớn nghĩa hai nút con giảm hỗn tạp nhiều nhất.
    \item Phép chọn tham lam được lặp lại ở từng nút.
  \end{itemize}
  \note{[Sources] Breiman et al., 1984.\\
  Talk track (0:40): Đọc công thức từ trái sang phải: impurity của nút cha trừ impurity có trọng số của hai nút con. Chọn feature và threshold có gain lớn nhất.}
\end{frame}

\begin{frame}
  \frametitle{Huấn luyện cây là một quá trình đệ quy}
  \chaptertag{3}{KIẾN THỨC NỀN TẢNG}
  \begin{tcolorbox}[focusbox,width=0.84\textwidth,top=3mm,bottom=3mm]
    {\small\ttfamily
    BuildTree($S$):\\
    \quad if Stop($S$): return Leaf($\hat p(y\mid S)$)\\
    \quad $(j^\star,t^\star)\leftarrow\arg\max_{j,t}\Delta G(j,t)$\\
    \quad $S_L,S_R\leftarrow$ Split($S,j^\star,t^\star$)\\
    \quad return Node($j^\star,t^\star$,\\
    \qquad BuildTree($S_L$), BuildTree($S_R$))}
  \end{tcolorbox}
  \vspace{0.28cm}
  \centering
  \textcolor{accentorange}{\textbf{Điểm dễ overfit:}} điều kiện dừng quá lỏng khiến quá trình tiếp tục đến các nhánh rất ít mẫu.
  \note{[Sources] Breiman et al., 1984; scikit-learn DecisionTreeClassifier documentation.\\
  Talk track (0:40): Theo dõi một lời gọi BuildTree: kiểm tra dừng, tìm split tốt nhất, chia dữ liệu và gọi lại cho hai nhánh. Overfitting xuất hiện khi điều kiện dừng cho phép cây đi quá sâu.}
\end{frame}

\begin{frame}
  \frametitle{Pre-pruning dừng sớm; CCP cắt cây sau khi fit}
  \chaptertag{5}{THỰC NGHIỆM VÀ PHÂN TÍCH KẾT QUẢ}
  \begin{columns}[T,onlytextwidth]
    \column{0.46\textwidth}
      \textbf{Pre-pruning}
      \begin{itemize}
        \item Giới hạn \texttt{max\_depth}.
        \item Yêu cầu số mẫu tối thiểu.
        \item Rủi ro: dừng quá sớm gây underfit.
      \end{itemize}
    \column{0.50\textwidth}
      \textbf{Cost-Complexity Pruning}
      \[
        T_\alpha=\arg\min_{T\subseteq T_0}
        \left[R(T)+\alpha|\mathcal{L}(T)|\right]
      \]
      $\alpha$ lớn tạo cây nhỏ hơn.
  \end{columns}
  \vspace{0.35cm}
  \begin{tcolorbox}[focusbox,width=0.90\textwidth,top=2mm,bottom=2mm]
    \centering
    Cả hai phương pháp đều thay đổi \textbf{topology}.
  \end{tcolorbox}
  \note{[Sources] Breiman et al., 1984; scikit-learn minimal cost-complexity pruning documentation.\\
  Talk track (0:40): Pre-pruning ra quyết định sớm, CCP đánh giá subtree sau khi đã thấy cây đầy đủ. Mục tiêu CCP cộng lỗi và penalty theo số lá.}
\end{frame}

% 21 --------------------------------------------------------------------------
\begin{frame}
  \frametitle{Hierarchical Shrinkage làm mềm các nút ít mẫu}
  \chaptertag{5}{THỰC NGHIỆM VÀ PHÂN TÍCH KẾT QUẢ}
  \begin{tcolorbox}[focusbox,width=0.90\textwidth,top=4mm,bottom=4mm]
    \centering
    $\displaystyle
      \hat p_{HS}(v)=\hat p_{HS}(pa(v))+
      \frac{n_v}{n_v+\lambda}
      \left[\hat p(v)-\hat p(pa(v))\right]$
  \end{tcolorbox}
  \vspace{0.42cm}
  \begin{itemize}
    \item $n_v$ lớn: giữ nhiều thông tin của nút con.
    \item $n_v$ nhỏ hoặc $\lambda$ lớn: kéo mạnh hơn về nút cha.
    \item Split, depth và số lá không đổi.
  \end{itemize}
  \note{[Sources] Agarwal et al., Hierarchical Shrinkage, 2022; HS benchmark notebook.\\
  Talk track (0:45): Giải thích hệ số n trên n cộng lambda. Nút đông mẫu được tin nhiều; nút ít mẫu mượn thông tin từ cha để giảm phương sai.}
\end{frame}
