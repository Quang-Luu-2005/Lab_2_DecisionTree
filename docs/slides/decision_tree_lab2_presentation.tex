\begin{frame}
  \frametitle{CART chọn phép chia làm giảm Gini nhiều nhất}
  \chaptertag{3}{KIẾN THỨC NỀN TẢNG}
  \begin{columns}[T,onlytextwidth]
    \column{0.60\textwidth}
      \begin{tcolorbox}[focusbox,width=\linewidth,top=3mm,bottom=3mm]
        \centering
        $\displaystyle Gini(S)=1-\sum_{k=1}^{K}p_k^2$\\[0.32cm]
        $\displaystyle \Delta G(j,t)=Gini(S)-\frac{n_L}{n}Gini(S_L)-\frac{n_R}{n}Gini(S_R)$
      \end{tcolorbox}
    \column{0.36\textwidth}
      \small
      \textbf{Ký hiệu}
      \begin{itemize}
        \item $p_k$: tỷ lệ lớp $k$ tại nút.
        \item $j,t$: feature và threshold.
        \item $n_L,n_R$: số mẫu hai nhánh.
      \end{itemize}
  \end{columns}
  \vspace{0.24cm}
  \small
  $Gini=0$ nghĩa là nút thuần. CART chọn $(j^\star,t^\star)$ làm $\Delta G$ lớn nhất, tức hai nút con thuần hơn nhiều nhất theo trung bình có trọng số.
  \note{[Sources] Breiman et al., 1984.\\
  Talk track (0:45): Giải thích impurity và weighted gain; đây là toán học cốt lõi của baseline.}
\end{frame}

\begin{frame}
  \frametitle{Điều kiện dừng quyết định cây có phát triển quá sâu hay không}
  \chaptertag{3}{KIẾN THỨC NỀN TẢNG}
  \begin{tcolorbox}[focusbox,width=0.84\textwidth,top=3mm,bottom=3mm]
    {\small\ttfamily
    BuildTree($S$):\\
    \quad if Stop($S$): return Leaf($\hat p(y\mid S)$)\\
    \quad $(j^\star,t^\star)\leftarrow\arg\max_{j,t}\Delta G(j,t)$\\
    \quad $S_L,S_R\leftarrow$ Split($S,j^\star,t^\star$)\\
    \quad return Node(BuildTree($S_L$), BuildTree($S_R$))}
  \end{tcolorbox}
  \vspace{0.28cm}
  \begin{tcolorbox}[riskbox,width=0.88\textwidth,top=2mm,bottom=2mm]
    \centering
    Stop quá lỏng $\Rightarrow$ nhánh ít mẫu $\Rightarrow$ prediction cực đoan và phương sai cao.
  \end{tcolorbox}
  \note{[Sources] Breiman et al., 1984; scikit-learn DecisionTreeClassifier documentation.\\
  Talk track (0:40): Đi qua pseudocode và chỉ ra vị trí xuất hiện overfitting.}

\begin{frame}
  \frametitle{Macro-F1 buộc mô hình quan tâm ngang nhau tới từng lớp}
  \chaptertag{4}{PHƯƠNG PHÁP NGHIÊN CỨU}
  \footnotesize
  \begin{columns}[T,onlytextwidth]
    \column{0.49\textwidth}
      \textbf{Precision và recall của lớp $k$}
      \begin{align*}
        P_k &= \frac{TP_k}{TP_k+FP_k},\\[-0.08cm]
        R_k &= \frac{TP_k}{TP_k+FN_k}.
      \end{align*}
      Precision: tỷ lệ dự đoán lớp $k$ là đúng.\\[0.08cm]
      Recall: tỷ lệ mẫu thật lớp $k$ được tìm thấy.
    \column{0.47\textwidth}
      \textbf{Cân bằng rồi trung bình theo lớp}
      \begin{align*}
        F1_k &= \frac{2P_kR_k}{P_k+R_k},\\[-0.08cm]
        Macro\text{-}F1 &= \frac{1}{K}\sum_{k=1}^{K}F1_k.
      \end{align*}
      Mỗi lớp đóng góp $1/K$, không phụ thuộc support lớn hay nhỏ.
  \end{columns}
  \vspace{0.18cm}
  \centering
  \textbf{Covertype dùng Macro-F1 để chọn cấu hình và báo cáo thêm recall theo lớp.}
  \note{[Sources] Precision, recall, F1 and Macro-F1 definitions in report Chapter 3.\\
  Talk track (0:50): Giải thích TP, FP, FN; nhấn mạnh mỗi lớp có trọng số bằng nhau trong Macro-F1.}
\end{frame}

% CCP mechanism --------------------------------------------------------------
\begin{frame}
  \frametitle{CCP cắt subtree sau khi fit cây đầy đủ}
  \chaptertag{5}{THỰC NGHIỆM VÀ PHÂN TÍCH KẾT QUẢ}
  \begin{tcolorbox}[focusbox,width=0.88\textwidth,top=3mm,bottom=3mm]
    \centering
    $\displaystyle T_{\alpha}=\arg\min_{T\subseteq T_0}
    \left[R(T)+\alpha|\mathcal{L}(T)|\right]$
  \end{tcolorbox}
  \vspace{0.28cm}
  \small
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \begin{itemize}
        \item $T_0$: cây đầy đủ ban đầu.
        \item $T$: một subtree của $T_0$.
        \item $R(T)$: empirical risk của cây $T$.
      \end{itemize}
    \column{0.48\textwidth}
      \begin{itemize}
        \item $|\mathcal{L}(T)|$: số lá.
        \item $\alpha$: mức phạt độ phức tạp.
        \item $\alpha$ lớn hơn $\Rightarrow$ ưu tiên cây ít lá hơn.
      \end{itemize}
  \end{columns}
  \vspace{0.10cm}
  \centering
  Khác pre-pruning, CCP nhìn thấy cây đã fit rồi mới quyết định nhánh nào nên loại.
  \note{[Sources] Breiman et al., 1984; scikit-learn cost-complexity pruning documentation.\\
  Talk track (0:50): Giải thích từng ký hiệu và sự khác biệt giữa post-pruning với early stopping.}
\end{frame}

% Hierarchical Shrinkage mechanism -------------------------------------------
\begin{frame}
  \frametitle{HS làm mềm dự đoán nhưng giữ nguyên topology}
  \chaptertag{5}{THỰC NGHIỆM VÀ PHÂN TÍCH KẾT QUẢ}
  \begin{tcolorbox}[focusbox,width=0.91\textwidth,top=3mm,bottom=3mm]
    \centering
    $\displaystyle \hat p_{HS}(v)=\hat p_{HS}(pa(v))+
    \frac{n_v}{n_v+\lambda}\left[\hat p(v)-\hat p(pa(v))\right]$
  \end{tcolorbox}
  \vspace{0.24cm}
  \small
  \begin{columns}[T,onlytextwidth]
    \column{0.49\textwidth}
      \begin{itemize}
        \item $v$: nút đang xét; $pa(v)$: nút cha.
        \item $\hat p(v)$: phân bố lớp ước lượng tại $v$.
        \item $n_v$: số mẫu train đi tới nút $v$.
      \end{itemize}
    \column{0.47\textwidth}
      \begin{itemize}
        \item $\lambda=0$: không shrink.
        \item $n_v$ nhỏ hoặc $\lambda$ lớn: dự đoán bị kéo mạnh về nút cha.
        \item Split, depth và số lá không thay đổi.
      \end{itemize}
  \end{columns}
  \note{[Sources] Agarwal et al., Hierarchical Shrinkage, 2022; method description in report Chapter 5.\\
  Talk track (0:55): Giải thích factor n/(n+lambda) trước khi trình bày kết quả E3.}
\end{frame}
