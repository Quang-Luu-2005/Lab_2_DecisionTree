\begin{frame}
  \frametitle{CART chọn phép chia làm giảm Gini nhiều nhất}
  \chaptertag{3}{KIẾN THỨC NỀN TẢNG}
  \begin{tcolorbox}[focusbox,width=0.90\textwidth,top=3mm,bottom=3mm]
    \centering
    $\displaystyle Gini(S)=1-\sum_{k=1}^{K}p_k^2$\\[0.32cm]
    $\displaystyle \Delta G(j,t)=Gini(S)-\frac{n_L}{n}Gini(S_L)-\frac{n_R}{n}Gini(S_R)$
  \end{tcolorbox}
  \vspace{0.38cm}
  \begin{itemize}
    \item $Gini=0$ khi nút chỉ chứa một lớp.
    \item Chọn $(j^\star,t^\star)=\arg\max_{j,t}\Delta G(j,t)$.
    \item Lặp cùng phép chọn tham lam tại mỗi nút con.
  \end{itemize}
  \note{[Sources] Breiman et al., 1984.\\
  Talk track (0:45): Giải thích impurity và weighted gain; đây là toán học cốt lõi của baseline.}
\end{frame}

\begin{frame}
  \frametitle{Điều kiện dừng quyết định cây có phát triển quá sâu hay không}
  \chaptertag{3}{KIẾN THỨC NỀN TẢNG}
  \begin{tcolorbox}[focusbox,width=0.84\textwidth,top=3mm,bottom=3mm]
    {\small\ttfamily
    BuildTree($S$):\\
    \quad if Stop($S$): return Leaf($\hat p(y\mid S)$)\\
    \quad $(j^\star,t^\star)\leftarrow\arg\max_{j,t}\Delta G(j,t)$\\
    \quad $S_L,S_R\leftarrow$ Split($S,j^\star,t^\star$)\\
    \quad return Node(BuildTree($S_L$), BuildTree($S_R$))}
  \end{tcolorbox}
  \vspace{0.28cm}
  \begin{tcolorbox}[riskbox,width=0.88\textwidth,top=2mm,bottom=2mm]
    \centering
    Stop quá lỏng $\Rightarrow$ nhánh ít mẫu $\Rightarrow$ prediction cực đoan và phương sai cao.
  \end{tcolorbox}
  \note{[Sources] Breiman et al., 1984; scikit-learn DecisionTreeClassifier documentation.\\
  Talk track (0:40): Đi qua pseudocode và chỉ ra vị trí xuất hiện overfitting.}
\end{frame}
